### 3. JavaScript Functions

```javascript
// View Toggle
function switchToCardView() {
    $('.employee-grid').show();
    $('#employeeTableView').hide();
    $('#btn-card-view').addClass('active');
    $('#btn-list-view').removeClass('active');
    localStorage.setItem('employeeViewPreference', 'card');
}

function switchToListView() {
    $('.employee-grid').hide();
    $('#employeeTableView').show();
    $('#btn-card-view').removeClass('active');
    $('#btn-list-view').addClass('active');
    localStorage.setItem('employeeViewPreference', 'list');
}

// Load preference
$(document).ready(function() {
    const savedView = localStorage.getItem('employeeViewPreference') || 'card';
    if (savedView === 'list') {
        switchToListView();
    }
});
```
